function(add_http_curl)
    add_compile_definitions(MBEDTLS_USER_CONFIG_FILE="${PROJECT_SOURCE_DIR}/core/libs/prx/libSceSsl/MbedConfig.hpp")
    set(CARES_STATIC ON CACHE BOOL "" FORCE)
    set(CARES_SHARED OFF CACHE BOOL "" FORCE)
    set(CARES_STATIC_PIC ON CACHE BOOL "" FORCE)
    set(CARES_INSTALL OFF CACHE BOOL "" FORCE)
    set(CARES_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(CARES_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    add_subdirectory(${PROJECT_SOURCE_DIR}/3rdparty/c-ares ${CMAKE_BINARY_DIR}/http-cares EXCLUDE_FROM_ALL)
    set(CARES_INCLUDE_DIR ${PROJECT_SOURCE_DIR}/3rdparty/c-ares/include)
    set(CARES_LIBRARY c-ares::cares_static)
    set(CARES_USE_STATIC_LIBS ON)
    set(ENABLE_ARES ON)
    set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
    set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory(${PROJECT_SOURCE_DIR}/3rdparty/zlib ${CMAKE_BINARY_DIR}/http-zlib EXCLUDE_FROM_ALL)
    set_target_properties(zlibstatic PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(ZLIB::ZLIB INTERFACE IMPORTED)
    set_target_properties(ZLIB::ZLIB PROPERTIES INTERFACE_LINK_LIBRARIES zlibstatic
        INTERFACE_INCLUDE_DIRECTORIES "${PROJECT_SOURCE_DIR}/3rdparty/zlib;${CMAKE_BINARY_DIR}/http-zlib")
    set(ZLIB_INCLUDE_DIR ${PROJECT_SOURCE_DIR}/3rdparty/zlib)
    set(ZLIB_LIBRARY zlibstatic)
    set(MBEDTLS_INCLUDE_DIR ${PROJECT_SOURCE_DIR}/3rdparty/mbedtls/include)
    set(MBEDTLS_LIBRARY mbedtls)
    set(MBEDX509_LIBRARY mbedx509)
    set(MBEDCRYPTO_LIBRARY mbedcrypto)
    set(HAVE_MBEDTLS_DES_CRYPT_ECB 1)
    set(BUILD_TESTING OFF)
    set(BUILD_CURL_EXE OFF)
    set(BUILD_SHARED_LIBS OFF)
    set(BUILD_STATIC_LIBS ON)
    set(CURL_DISABLE_INSTALL ON)
    set(CURL_ENABLE_SSL ON)
    set(CURL_USE_MBEDTLS ON)
    set(CURL_USE_OPENSSL OFF)
    set(CURL_CA_BUNDLE "none" CACHE STRING "Use guest certificate authorities" FORCE)
    set(CURL_CA_PATH "none" CACHE STRING "Use guest certificate authorities" FORCE)
    set(CURL_ZLIB ON CACHE STRING "Build curl with bundled zlib" FORCE)
    set(CURL_BROTLI OFF CACHE STRING "Build curl without external brotli" FORCE)
    set(CURL_ZSTD OFF CACHE STRING "Build curl without external zstd" FORCE)
    set(USE_LIBIDN2 OFF)
    set(CURL_USE_LIBPSL OFF)
    set(USE_NGHTTP2 OFF)
    set(USE_NGHTTP3 OFF)
    set(CURL_USE_LIBSSH2 OFF)
    set(CURL_USE_LIBSSH OFF)
    set(CURL_USE_GSSAPI OFF)
    set(CURL_DISABLE_LDAP ON)
    set(CURL_DISABLE_LDAPS ON)
    set(CURL_DISABLE_SMB ON)
    set(CURL_WINDOWS_SSPI OFF)
    set(CURL_USE_PKGCONFIG OFF)
    set(CURL_USE_CMAKECONFIG OFF)
    add_subdirectory(${PROJECT_SOURCE_DIR}/3rdparty/curl ${CMAKE_BINARY_DIR}/http-curl EXCLUDE_FROM_ALL)
endfunction()
add_http_curl()

add_library(http_tls_adapter STATIC EXCLUDE_FROM_ALL src/TlsAdapter.cpp)
target_link_libraries(http_tls_adapter PRIVATE mbedtls)
set_target_properties(http_tls_adapter PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(libSceHttp SHARED EXCLUDE_FROM_ALL Export.cpp src/Parse.cpp src/Uri.cpp src/Transport.cpp)
target_link_libraries(libSceHttp PRIVATE CURL::libcurl_static http_tls_adapter mbedtls mbedx509 mbedcrypto libSceSsl libSceNetCtl)
target_link_options(libSceHttp PRIVATE LINKER:--wrap=mbedtls_ssl_set_hostname)
if(MINGW OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(libSceHttp PRIVATE LINKER:--exclude-libs,ALL)
endif()
install(FILES ${PROJECT_SOURCE_DIR}/3rdparty/curl/COPYING DESTINATION licenses/curl)
install(FILES ${PROJECT_SOURCE_DIR}/3rdparty/zlib/LICENSE DESTINATION licenses/zlib)
install(FILES ${PROJECT_SOURCE_DIR}/3rdparty/c-ares/LICENSE.md DESTINATION licenses/c-ares)
target_include_directories(libSceHttp PRIVATE ${LIBS_INCLUDE_DIR})
configure_windows_unwind(libSceHttp)
